   126→  await testSection('2. Runtime: Stop-during-TTS releases wait immediately', async () => {

   128→
   129→    // Mirror NexVoiceConversation's waitForTtsPlayback + releaseTtsPlaybackWait
   130→    // + abortCurrentTurn semantics with a fake implementation.
   131→    let currentTtsRequestId = 0;
   132→    let ttsPlaybackResolve: (() => void) | null = null;
   133→    let ttsPlaybackRequestId: number | null = null;
   134→    let ttsPlaybackTimeout: ReturnType<typeof setTimeout> | null = null;
   135→
   136→    function releaseTtsPlaybackWait() {
   137→      if (ttsPlaybackResolve) {
   138→        const resolve = ttsPlaybackResolve;
   139→        ttsPlaybackResolve = null;
   140→        ttsPlaybackRequestId = null;
   141→        if (ttsPlaybackTimeout) { clearTimeout(ttsPlaybackTimeout); ttsPlaybackTimeout = null; }
   142→        resolve();
   143→      }
   144→    }
   145→
   146→    function waitForTtsPlayback(requestId: number): Promise<void> {
   147→      if (currentTtsRequestId !== requestId) return Promise.resolve();
   148→      releaseTtsPlaybackWait();
   149→      return new Promise<void>((resolve) => {
   150→        ttsPlaybackRequestId = requestId;
   151→        ttsPlaybackResolve = resolve;
   152→        ttsPlaybackTimeout = setTimeout(() => releaseTtsPlaybackWait(), 30000);
   153→      });
   154→    }
   155→
   156→    function abortCurrentTurn() {
   157→      currentTtsRequestId++;
   158→      releaseTtsPlaybackWait();
   159→    }
   160→
   161→    // Scenario: speakResponse is awaiting waitForTtsPlayback(1).
   162→    // User clicks Stop → abortCurrentTurn() fires.
   163→    currentTtsRequestId = 1;
   164→    const waitPromise = waitForTtsPlayback(1);
   165→
   166→    // Yield to let the promise set up
   167→    await new Promise((r) => setTimeout(r, 10));
   168→
   169→    // The wait should still be pending (no signal yet)
   170→    let resolved = false;
   171→    waitPromise.then(() => { resolved = true; });
   172→    await new Promise((r) => setTimeout(r, 10));
   173→    assert(resolved === false, 'waitForTtsPlayback still pending before Stop');
   174→
   175→    // Now simulate Stop → abortCurrentTurn
   176→    const stopStart = Date.now();
   177→    abortCurrentTurn();
   178→    await waitPromise;
   179→    const elapsed = Date.now() - stopStart;
   180→
   181→    assert(resolved === true, 'waitForTtsPlayback resolved after Stop');
   182→    assert(elapsed < 1000, `waitForTtsPlayback resolved immediately (elapsed=${elapsed}ms, not 30s)`);

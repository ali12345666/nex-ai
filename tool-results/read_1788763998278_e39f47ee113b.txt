     1→/**
     2→ * NEX AI — Voice Service (Phase 30 + Phase 116 JARVIS)
     3→ *
     4→ * Always-ready voice system with Web Audio API microphone analysis +
     5→ * browser SpeechRecognition for STT + SpeechSynthesis for TTS.
     6→ *

     8→ *   - Continuous Conversation mode (auto-restart listening after TTS)

    22→ *       ├── Wake Word: regex match on partial transcripts
    23→ *       └── state: state machine → callbacks → VoiceController
    24→ */

   157→  async enableMicrophone(): Promise<boolean> {
   158→    if (this._stream) return true;
   159→    try {
   160→      this._micPermission = null; // pending
   161→      console.log(`[VOICE] calling getUserMedia...`);
   162→      this._stream = await navigator.mediaDevices.getUserMedia({
   163→        audio: { echoCancellation: true, noiseSuppression: true, autoGainControl: true },
   164→      });
   165→      this._micPermission = true;
   166→      this.callbacks.onPermissionChange?.(true);
   167→      console.log(`[VOICE] getUserMedia resolved — stream tracks: ${this._stream.getTracks().length}`);
   168→
   169→      this._audioContext = new AudioContext();
   170→      console.log(`[VOICE] AudioContext created — state: ${this._audioContext.state}`);
   171→
   172→      // CRITICAL: AudioContext starts in 'suspended' state in Electron.
   173→      if (this._audioContext.state === 'suspended') {
   174→        console.log(`[VOICE] AudioContext suspended — calling resume()...`);
   175→        try {
   176→          await this._audioContext.resume();
   177→          console.log(`[VOICE] AudioContext resumed — state: ${this._audioContext.state}`);
   178→        } catch (err: any) {
   179→          console.warn(`[VOICE] AudioContext resume failed: ${err?.message}`);
   180→        }
   181→      }
   182→
   183→      const source = this._audioContext.createMediaStreamSource(this._stream);
   184→      this._analyser = this._audioContext.createAnalyser();
   185→      this._analyser.fftSize = 256;
   186→      this._analyser.smoothingTimeConstant = 0.8;
   187→      source.connect(this._analyser);
   188→      this._dataArray = new Uint8Array(new ArrayBuffer(this._analyser.frequencyBinCount));
   189→
   190→      // ── PCM audio capture for LocalVoiceEngine ──────────────────────────
   191→      this._scriptProcessor = this._audioContext.createScriptProcessor(4096, 1, 1);
   192→      console.log(`[VOICE] ScriptProcessorNode created — bufferSize: ${this._scriptProcessor.bufferSize}`);
   193→
   194→      this._audioProcessEventCount = 0;
   195→      this._scriptProcessor.onaudioprocess = (event: AudioProcessingEvent) => {
   196→        this._audioProcessEventCount++;
   197→        if (!this._ipcFeedingEnabled) {
   198→          if (this._audioProcessEventCount <= 3) {
   199→            console.log(`[VOICE] onaudioprocess (#${this._audioProcessEventCount}) but IPC feeding disabled`);
   200→          }
   201→          return;
   202→        }
   203→        const inputBuffer = event.inputBuffer;
   204→        const channelData = inputBuffer.getChannelData(0);
   205→
   206→        // Downsample from 48kHz to 16kHz
   207→        const downsampled = this.downsampleTo16k(channelData);
   208→
   209→        // Convert Float32 to Int16 PCM
   210→        const pcm16 = new Int16Array(downsampled.length);
   211→        for (let i = 0; i < downsampled.length; i++) {
   212→          const s = Math.max(-1, Math.min(1, downsampled[i]));
   213→          pcm16[i] = s < 0 ? s * 0x8000 : s * 0x7FFF;
   214→        }
   215→
   216→        // Send the PCM chunk to the main process via IPC
   217→        const chunkBuffer = pcm16.buffer;
   218→        this._chunksSent++;
   219→        this._lastChunkSize = chunkBuffer.byteLength;
   220→        try {
   221→          const ipcAvailable = !!(window as any).nexAPI?.voiceFeedAudioChunk;
   222→          if (!ipcAvailable) {
   223→            if (this._chunksSent <= 3) {
   224→              console.warn(`[VOICE] voiceFeedAudioChunk NOT available on window.nexAPI!`);
   225→            }
   226→            return;
   227→          }
   228→          window.nexAPI!.voiceFeedAudioChunk!(chunkBuffer);
   229→        } catch (err: any) {
   230→          if (this._chunksSent <= 3) {
   231→            console.warn(`[VOICE] voiceFeedAudioChunk error: ${err?.message}`);
   232→          }
   233→        }
   234→
   235→        // Compute audio level (RMS) for VAD + Orb animation
   236→        let sum = 0;
   237→        for (let i = 0; i < downsampled.length; i++) {
   238→          sum += downsampled[i] * downsampled[i];
   239→        }
   240→        const rms = Math.sqrt(sum / downsampled.length);
   241→        const gated = rms < this.config.noiseFloor ? 0 : rms;
   242→        const normalized = Math.min(1, gated * 4);
   243→        const speed = normalized > this._smoothedLevel
   244→          ? this.config.attackSpeed : this.config.releaseSpeed;
   245→        this._smoothedLevel += (normalized - this._smoothedLevel) * speed;
   246→        this.callbacks.onAudioLevel?.(this._smoothedLevel);
   247→        try {
   248→          window.nexAPI?.voiceFeedAudioLevel?.(this._smoothedLevel);
   249→        } catch { /* best-effort */ }
   250→
   251→        // Phase 116: VAD (Voice Activity Detection) — detect speech start/end
   252→        this.processVAD(normalized);
   253→      };
   254→      source.connect(this._scriptProcessor);
   255→      this._scriptProcessor.connect(this._audioContext.destination);
   256→      console.log(`[VOICE] ScriptProcessorNode connected — source→processor→destination`);
   257→
   258→      this.startAudioLoop();
   259→      return true;
   260→    } catch (err: any) {
   261→      this._micPermission = false;
   262→      this.callbacks.onPermissionChange?.(false);
   263→      const msg = err.name === 'NotAllowedError' ? 'Microphone access denied'
   264→        : err.name === 'NotFoundError' ? 'No microphone found'
   265→        : `Microphone error: ${err.message}`;
   266→      console.error(`[VOICE] enableMicrophone failed: ${msg} (name=${err.name})`);
   267→      this.callbacks.onError?.(msg);
   268→      return false;
   269→    }
   270→  }

   272→  /**
   273→   * Phase 116: Voice Activity Detection — detects speech start and end
   274→   * based on audio level. When speech ends (silence after speech), we
   275→   * know the user finished their sentence and can process the transcript.
   276→   *
   277→   * Phase 18 Stage 3 (AUDIO-NO-MUTE-TTS fix): skip VAD processing when the
   278→   * engine is in 'speaking' state. This prevents TTS audio bleed from
   279→   * updating the `_vadState` (harmless but wasteful, and prevents stale
   280→   * 'speech' state from persisting after TTS ends). The main-side VAD
   281→   * handles barge-in detection with its own higher threshold — the
   282→   * renderer VAD is purely for the renderer-side speech/silence tracking
   283→   * which is no longer used for any action (barge-in was removed in P2-6).
   284→   *
   285→   * The audio level is still sent to main unconditionally (via
   286→   * `voiceFeedAudioLevel`) so the main-side VAD can detect barge-in. Only
   287→   * the renderer-side `processVAD` computation is skipped.
   288→   */
   289→  private processVAD(level: number): void {
   290→    // Phase 18 Stage 3: skip when engine is speaking (TTS in progress).
   291→    if (this._stateConditions.get('engine') === 'speaking') return;
   292→
   293→    const now = Date.now();
   294→    const isLoud = level > this.config.vadSilenceThreshold;
   295→
   296→    if (isLoud) {
   297→      // Speech detected
   298→      if (this._vadState === 'silence') {
   299→        this._vadState = 'speech';
   300→        this._vadSpeechStart = now;
   301→        this._vadSilenceStart = 0;
   302→        // Phase 18 (P2-6 fix): REMOVED the renderer-side barge-in branch.
   303→        // Previously: `if (this._ttsActive && this._bargeInEnabled) {
   304→        //   this.stopSpeaking(); if (this._mode === 'continuous' && !this._sttActive)
   305→        //   { this.startSTT(); this.setCondition('mic', 'listening'); ... } }`
   306→        // This was dead code — `_ttsActive` was never set to true (no caller
   307→        // invoked voiceService.speak()). The renderer VAD cannot distinguish
   308→        // user speech from TTS audio bleed (browser echo cancellation is
   309→        // imperfect with speakers). Barge-in detection should be on the
   310→        // MAIN side (where sttActive gates VAD) — see Phase 18 audit P1-1
   311→        // for the recommended main-side barge-in fix (Stage 3, not yet
   312→        // implemented). For now, the renderer VAD only transitions state —
   313→        // it does NOT trigger barge-in. This is safe because the main-side
   314→        // VAD (local-voice-engine.ts:199) is gated by `sttActive` and
   315→        // `ttsActive` (engine.speak sets ttsActive=true during synthesis),
   316→        // so the main side correctly suppresses transcription during TTS.
   317→      }
   318→    } else {
   319→      // Silence detected
   320→      if (this._vadState === 'speech') {
   321→        if (this._vadSilenceStart === 0) {
   322→          this._vadSilenceStart = now;
   323→        }
   324→        // If silence lasts long enough, declare speech ended
   325→        if (now - this._vadSilenceStart >= this.config.vadSilenceDurationMs) {
   326→          this._vadState = 'silence';
   327→          console.log('[VOICE] VAD: speech ended (silence detected)');
   328→        }
   329→      }
   330→    }
   331→  }

   333→  /**
   334→   * Downsample Float32 audio from 48kHz to 16kHz by taking every 3rd sample.
   335→   */
   336→  private downsampleTo16k(input: Float32Array): Float32Array {
   337→    const ratio = 3; // 48000 / 16000
   338→    const outputLength = Math.floor(input.length / ratio);
   339→    const output = new Float32Array(outputLength);
   340→    for (let i = 0; i < outputLength; i++) {
   341→      output[i] = input[i * ratio];
   342→    }
   343→    return output;
   344→  }

   457→  private recomputeState(): void {
   458→    let newState: VoiceState = 'idle';
   459→    let highest = 0;
   460→    for (const state of this._stateConditions.values()) {
   461→      const p = STATE_PRIORITY[state] || 0;
   462→      if (p > highest) { highest = p; newState = state; }
   463→    }
   464→    if (newState !== this._state) {
   465→      // Phase 18 (BUG-37): enforce the Orb state machine. Validate the
   466→      // transition via `safeOrbTransition()`. If the transition is invalid
   467→      // per the relaxed VALID_TRANSITIONS graph in orb-state.ts, the
   468→      // function returns the CURRENT state (blocking the invalid transition)
   469→      // and logs `[ORB_STATE] Invalid transition blocked: <from> → <to>`.
   470→      //
   471→      // This prevents invalid transitions (e.g. offline → speaking without
   472→      // going through idle) while allowing all documented production flows
   473→      // (voice, chat, agent, voice-agent, barge-in, flash-state recovery).
   474→      // See orb-state.ts VALID_TRANSITIONS comment for the full list of
   475→      // relaxed transitions.
   476→      //
   477→      // Cast VoiceState ↔ NexOrbState: VoiceState has 9 states (a subset of
   478→      // NexOrbState's 13), so the cast is safe — every VoiceState is a
   479→      // valid NexOrbState. The returned NexOrbState is similarly castable
   480→      // back to VoiceState (the blocked case returns the current state,
   481→      // which was already a VoiceState).
   482→      const validated = safeOrbTransition(this._state as NexOrbState, newState as NexOrbState) as VoiceState;
   483→      if (validated !== this._state) {
   484→        this._state = validated;
   485→        this.callbacks.onStateChange?.(validated);
   486→      }
   487→      // If validated === this._state, the transition was blocked — the
   488→      // warning was already logged by safeOrbTransition. We keep the
   489→      // current state (do NOT update this._state, do NOT fire callback).
   490→    }
   491→  }
   492→
   493→  private startAudioLoop(): void {
   494→    if (this._rafId !== null) return;
   495→    const loop = () => {
   496→      if (!this._analyser || !this._dataArray) return;
   497→      this._analyser.getByteTimeDomainData(this._dataArray);
   498→      let sum = 0;
   499→      for (let i = 0; i < this._dataArray.length; i++) {
   500→        const val = (this._dataArray[i] - 128) / 128;
   501→        sum += val * val;
   502→      }
   503→      const rms = Math.sqrt(sum / this._dataArray.length);
   504→      const gated = rms < this.config.noiseFloor ? 0 : rms;
   505→      const normalized = Math.min(1, gated * 4);
   506→      const speed = normalized > this._smoothedLevel
   507→        ? this.config.attackSpeed : this.config.releaseSpeed;
   508→      this._smoothedLevel += (normalized - this._smoothedLevel) * speed;
   509→      this.callbacks.onAudioLevel?.(this._smoothedLevel);
   510→      this._audioLevelLogCount++;
   511→      if (this._audioLevelLogCount % 60 === 0) {
   512→        console.log(`[ORB_AUDIO] VoiceService: rms=${rms.toFixed(4)} smoothed=${this._smoothedLevel.toFixed(4)}`);
   513→      }
   514→      this._rafId = requestAnimationFrame(loop);
   515→    };
   516→    this._rafId = requestAnimationFrame(loop);
   517→  }
   518→
   519→  private stopAudioLoop(): void {
   520→    if (this._rafId !== null) {
   521→      cancelAnimationFrame(this._rafId);
   522→      this._rafId = null;
   523→    }
   524→  }
   525→
   526→  private startSTT(): void {
   527→    const SR = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition;
   528→    if (!SR) {
   529→      // Phase 116: Browser SpeechRecognition is NOT available in Electron.
   530→      // Use the main-side whisper STT path instead.
   531→      // The whisper path: mic → IPC → whisper → transcript → voice-conversation-user IPC
   532→      console.log('[VOICE] Browser STT not available — using main-side whisper STT');
   533→      this._sttActive = true;
   534→      this.setCondition('mic', 'listening');
   535→      return;
   536→    }
   537→    this.stopSTT();
   538→    const recognition = new SR();
   539→    recognition.continuous = true;
   540→    recognition.interimResults = true;
   541→    recognition.lang = this.config.language;
   542→    recognition.onresult = (event: any) => {
   543→      let interim = '';
   544→      let final = '';
   545→      for (let i = event.resultIndex; i < event.results.length; i++) {
   546→        const result = event.results[i];
   547→        if (result.isFinal) final += result[0].transcript;
   548→        else interim += result[0].transcript;
   549→      }
   550→      if (interim) {
   551→        // Phase 116: Check for wake word in interim results
   552→        this.checkWakeWord(interim);
   553→        this.callbacks.onPartialTranscript?.(interim);
   554→      }
   555→      if (final) {
   556→        console.log(`[VOICE] browser STT transcript: "${final.trim().substring(0, 100)}"`);
   557→        // Phase 116: Check for wake word in final results
   558→        const wakeWordFound = this.checkWakeWord(final);
   559→        if (wakeWordFound) {
   560→          // Strip the wake word from the transcript
   561→          const stripped = final.trim().replace(
   562→            new RegExp(`\\b${this.config.wakeWord}\\b`, 'i'),
   563→            ''
   564→          ).trim();
   565→          if (stripped) {
   566→            this.callbacks.onFinalTranscript?.(stripped);
   567→          } else {
   568→            // Just the wake word — signal that NEX should respond "بله؟"
   569→            this.callbacks.onWakeWord?.();
   570→          }
   571→        } else if (this._wakeWordDetected || this._mode === 'continuous') {
   572→          // In continuous mode or after wake word, send all transcripts
   573→          this.callbacks.onFinalTranscript?.(final.trim());
   574→        }
   575→        this._wakeWordDetected = wakeWordFound;
   576→      }
   577→    };
   578→    recognition.onerror = (event: any) => {
   579→      console.warn(`[VOICE] browser STT error: ${event.error}`);
   580→      if (event.error === 'not-allowed') {
   581→        this.callbacks.onError?.('Speech recognition permission denied');
   582→        this._shouldRestartSTT = false;
   583→      }
   584→    };
   585→    recognition.onend = () => {
   586→      this._sttActive = false;
   587→      if (this._mode === 'continuous' && this._shouldRestartSTT) {
   588→        setTimeout(() => {
   589→          if (this._mode === 'continuous' && this._shouldRestartSTT) this.startSTT();
   590→        }, 100);
   591→      }
   592→    };
   593→    try {
   594→      recognition.start();
   595→      this._recognition = recognition;
   596→      this._sttActive = true;
   597→      console.log('[VOICE] browser STT started');
   598→    } catch { /* already started */ }
   599→  }
   600→
   601→  /**
   602→   * Phase 116: Check if the wake word ("NEX") appears in the transcript.
   603→   * If found, sets _wakeWordDetected = true and calls onWakeWord callback.
   604→   * Returns true if wake word was found.
   605→   */
   606→  private checkWakeWord(text: string): boolean {
   607→    const lower = text.toLowerCase().trim();
   608→    const wakeWord = this.config.wakeWord.toLowerCase();
   609→
   610→    // Check if the wake word appears as a standalone word
   611→    const regex = new RegExp(`\\b${wakeWord}\\b`, 'i');
   612→    if (regex.test(lower)) {
   613→      console.log(`[VOICE] Wake word detected: "${this.config.wakeWord}"`);
   614→      this._wakeWordDetected = true;
   615→      this.callbacks.onWakeWord?.();
   616→      return true;
   617→    }
   618→    return false;
   619→  }
   620→
   621→  private stopSTT(): void {
   622→    this._shouldRestartSTT = false;
   623→    this._sttActive = false;
   624→    if (this._recognition) {
   625→      try { this._recognition.stop(); } catch { /* already stopped */ }
   626→      this._recognition = null;
   627→    }
   628→  }
   629→}

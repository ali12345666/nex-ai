   199→  // ── Phase 6: Background Task Queue → Orb wiring ─────────────────────────────
   200→  // Listen for task-queue events from main and drive the Orb via a NEW

   204→  // The Orb state priority resolution in voice-service.ts picks the highest

   209→  // Mapping (mirrors main-side tasks/orb-bridge.ts):
   210→  //   task_started/progress → 'working'
   211→  //   task_completed        → 'success' (brief flash, then clear after 1.5s)
   212→  //   task_failed/recovered → 'error'   (brief flash, then clear after 1.5s)

   218→    // Phase 18 (P2-1): Stale-timer protection for 'queue' condition
   219→    // auto-clears. Previously `queueTimers` accumulated ALL timer IDs and
   220→    // only cleared them on unmount — a stale timer from a completed/failed
   221→    // task could fire during a NEW task and clear the NEW task's condition
   222→    // (brief Orb blip from working → idle). Now we track ONLY the latest
   223→    // timer for the 'queue' condition and clear the previous one before
   224→    // scheduling a new one. The invariant is:
   225→    //   A stale timer from an old task must NEVER clear the condition
   226→    //   belonging to a newer task.
   227→    let queueClearTimer: ReturnType<typeof setTimeout> | null = null;
   228→    const scheduleQueueClear = () => {
   229→      // Clear any previously-scheduled timer for the 'queue' condition.
   230→      if (queueClearTimer !== null) {
   231→        clearTimeout(queueClearTimer);
   232→        queueClearTimer = null;
   233→      }
   234→      // Schedule the new 1500ms auto-clear, capturing the timer ID.
   235→      queueClearTimer = setTimeout(() => {
   236→        voiceController.clearCondition('queue');
   237→        queueClearTimer = null;
   238→      }, 1500);
   239→    };
   240→

   242→      const type = event?.type as string;
   243→      if (!type) return;
   244→
   245→      if (type === 'task_started' || type === 'task_progress') {
   246→        voiceController.setCondition('queue', 'working');
   247→      } else if (type === 'task_completed') {
   248→        voiceController.setCondition('queue', 'success');
   249→        // Phase 18 (P2-1): capture timer ID; clears previous 'queue' timer.
   250→        scheduleQueueClear();
   251→      } else if (type === 'task_failed' || type === 'task_recovered') {
   252→        voiceController.setCondition('queue', 'error');
   253→        scheduleQueueClear();
   254→      } else if (type === 'task_cancelled') {
   255→        voiceController.setCondition('queue', 'cancelled');
   256→        scheduleQueueClear();
   257→      }
   258→      // task_enqueued, task_paused, queue_state → no Orb change
   259→    });
   260→
   261→    return () => {
   262→      off();
   263→      // Phase 18 (P2-1): clear the pending timer on unmount.
   264→      if (queueClearTimer !== null) clearTimeout(queueClearTimer);
   265→    };
   266→  }, []);
   267→
   268→  // ── Bridge: main-side voice conversation state → Orb animation ────────────
   269→  // The LocalVoiceEngine (whisper + piper) emits 'voice-conversation-state'

   272→  // conversation states (listening/thinking/speaking) to the Orb so it

   274→  //
   275→  // NOTE: The voice-start-mic-capture / voice-stop-mic-capture listeners are
   276→  // registered in App.tsx (root level) to ensure they're ready before the
   277→  // main process sends events during startup.
   278→  useEffect(() => {

   280→    // auto-clear (voice-conversation-error). Track ONLY the latest timer
   281→    // and clear the previous one before scheduling a new one.
   282→    let engineClearTimer: ReturnType<typeof setTimeout> | null = null;
   283→    const scheduleEngineClear = () => {
   284→      if (engineClearTimer !== null) {
   285→        clearTimeout(engineClearTimer);
   286→        engineClearTimer = null;
   287→      }
   288→      engineClearTimer = setTimeout(() => {
   289→        voiceController.clearCondition('engine');
   290→        engineClearTimer = null;
   291→      }, 1500);
   292→    };

   294→    const off = window.nexAPI?.onVoiceConversationState?.((ev: any) => {
   295→      const state = ev?.state as string;
   296→      if (!state) return;
   297→      // Phase 18 Stage 4 (GAP-7 fix): default to 'engine' for backward-compat
   298→      // (engine sender already had source='engine'; conversation sender now
   299→      // sends source='conversation'). Both use the same orbStateMap.
   300→      const source = ev?.source || 'engine';
   301→      console.log(`[ORB_TRACE_RENDERER] incoming state=${state} source=${source}`);
   302→
   303→      // Map conversation/engine state to Orb state
   304→      // Phase 116 JARVIS: Extended with new states (working, success, cancelled)
   305→      const orbStateMap: Record<string, string> = {
   306→        idle: 'idle',
   307→        initializing: 'initializing',
   308→        ready: 'ready',
   309→        listening: 'listening',
   310→        thinking: 'thinking',
   311→        speaking: 'speaking',
   312→        working: 'working',
   313→        active: 'active',
   314→        success: 'success',
   315→        cancelled: 'cancelled',
   316→        interrupted: 'active',
   317→        error: 'error',
   318→      };
   319→      const orbState = orbStateMap[state] || 'idle';
   320→      console.log(`[ORB_TRACE_RENDERER] mapped orbState=${orbState}`);
   321→
   322→      // Phase 18 Stage 4 (GAP-7 fix): route to separate condition keys based
   323→      // on the source field. Previously both conversation and engine FSMs
   324→      // used the same 'engine' condition key, causing a race where engine
   325→      // 'idle' (from stopSpeaking) cleared the condition while the
   326→      // conversation was still 'speaking' (waiting for playback). Now:
   327→      //   - conversation FSM → 'conversation' condition key
   328→      //   - engine FSM → 'engine' condition key (unchanged)
   329→      // Both contribute to the Orb state via the priority system
   330→      // (STATE_PRIORITY in voice-service.ts). The higher-priority active
   331→      // condition wins. This structurally eliminates the race: engine
   332→      // clearing its condition does NOT clear the conversation's condition.
   333→      const conditionKey = source === 'conversation' ? 'conversation' : 'engine';
   334→
   335→      // Drive the voiceController's state machine with the main-side states.
   336→      // Phase 116: Map new states to VoiceState equivalents.
   337→      if (orbState === 'listening') {
   338→        voiceController.setCondition(conditionKey, 'listening');
   339→      } else if (orbState === 'thinking') {
   340→        voiceController.setCondition(conditionKey, 'thinking');
   341→      } else if (orbState === 'speaking') {
   342→        voiceController.setCondition(conditionKey, 'speaking');
   343→      } else if (orbState === 'working' || orbState === 'active') {
   344→        voiceController.setCondition(conditionKey, 'working');
   345→      } else if (orbState === 'error') {
   346→        voiceController.setCondition(conditionKey, 'error');
   347→      } else {
   348→        // idle / ready / success / cancelled / initializing — clear the
   349→        // source-specific condition key.
   350→        voiceController.clearCondition(conditionKey);
   351→      }
   352→
   353→      // Log the controller's resolved state for diagnostics
   354→      console.log(`[ORB_TRACE_CONTROLLER] conditions=${conditionKey}:${orbState} resolvedState=${voiceController.orbState}`);
   355→    });

   372→    });
   373→

   378→      if (!text || !text.trim()) return;

   380→      // The conversation system already speaks the response via TTS.
   381→      // We don't dispatch it to Chat — the conversation pipeline handles
   382→      // the full voice→AI→TTS loop independently of the text Chat panel.
   383→    });
   384→
   385→    // Phase 17 (VOICE-ERROR-IPC-NOLISTENER fix): subscribe to
   386→    // voice-conversation-error IPC so voice engine errors (Whisper model
   387→    // missing, Piper model missing, mic denied, etc.) drive the Orb to
   388→    // 'error' state instead of being silently dropped. Previously the
   389→    // only subscriber was VoiceCenterPanel (which is dead code — never
   390→    // imported). Now AppShell subscribes and flashes the Orb red for
   391→    // 1.5s, matching the agent/chat error UX.
   392→    const offError = window.nexAPI?.onVoiceConversationError?.((ev: any) => {
   393→      const message = ev?.message || 'Voice engine error';

   395→      voiceController.setCondition('engine', 'error');
   396→      // Phase 18 (P2-1): capture timer ID; clears previous 'engine' timer.
   397→      scheduleEngineClear();
   398→    });
   399→
   400→    return () => {
   401→      if (off) off();
   402→      if (offUser) offUser();
   403→      if (offNex) offNex();
   404→      if (offError) offError();
   405→      // Phase 18 (P2-1): clear the pending 'engine' timer on unmount.
   406→      if (engineClearTimer !== null) clearTimeout(engineClearTimer);
   407→    };
   408→  }, []);

   422→      case 'planner': return <Suspense fallback={<PanelLoading />}><PlannerPanel /></Suspense>;

   438→    >
   439→      {/* Main row: Nav + Center + Chat

   453→        <div
   454→          className="flex-1 relative overflow-hidden"

   460→          <div className="absolute inset-0 flex flex-col items-center justify-center pointer-events-none">
   461→            {/* UI-16 §13: Orb header — ONLY "NEX AI", minimal */}
   462→            {showOrb && (

   469→                  NEX AI

   473→
   474→            {/* Orb container — LARGE, responsive. Fills available central workspace. */}

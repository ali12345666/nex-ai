   223→      if (type === 'task_started' || type === 'task_progress') {
   224→        voiceController.setCondition('queue', 'working');

   226→        voiceController.setCondition('queue', 'success');

   230→        voiceController.setCondition('queue', 'error');
   231→        const t = window.setTimeout(() => voiceController.clearCondition('queue'), 1500);
   232→        queueTimers.push(t);
   233→      } else if (type === 'task_cancelled') {
   234→        voiceController.setCondition('queue', 'cancelled');
   235→        const t = window.setTimeout(() => voiceController.clearCondition('queue'), 1500);
   236→        queueTimers.push(t);
   237→      }
   238→      // task_enqueued, task_paused, queue_state → no Orb change
   239→    });

   242→      off();

    62→  // ── Phase 33: Auto-save conversation to persistence ──
    63→  const saveConversation = useCallback(async (msgs?: NexMessage[]) => {

    65→    if (toSave.length === 0) return; // don't save empty conversations

    70→
    71→    // Don't re-save if nothing meaningful changed since last save
    72→    const now = Date.now();

    76→    setConversationTitle(title);

    94→        title,
    95→        createdAt: conversationCreatedAtRef.current || now,

   100→        provider: aiMode,
   101→        model: activeLocalModel?.name,
   102→        mode: aiMode,
   103→      });

   105→      setLastSavedAt(now);
   106→    } catch { /* non-fatal: chat still works */ }
   107→  }, [projectPath, aiMode, activeLocalModel, lastSavedAt]);
   108→
   109→  const conversationCreatedAtRef = useRef<number | null>(null);
   110→
   111→  // ── Phase 33: Load / New conversation from external events ──

   125→            setConversationId(r.data.id);
   126→            setConversationTitle(r.data.title || '');
   127→            conversationIdRef.current = r.data.id;
   128→            conversationTitleRef.current = r.data.title || '';
   129→            conversationCreatedAtRef.current = r.data.createdAt || null;
   130→            setLastSavedAt(Date.now());
   131→          }
   132→        } catch { /* graceful */ }
   133→      }
   134→    };
   135→    const newHandler = () => {
   136→      setMessages([]);
   137→      setConversationId(null);
   138→      setConversationTitle('');
   139→      conversationIdRef.current = null;
   140→      conversationTitleRef.current = '';
   141→      conversationCreatedAtRef.current = null;
   142→      setLastSavedAt(null);
   143→      setError(null);
   144→    };
   145→    window.addEventListener('nex:load-conversation', loadHandler);
   146→    window.addEventListener('nex:new-conversation', newHandler);
   147→    return () => {
   148→      window.removeEventListener('nex:load-conversation', loadHandler);
   149→      window.removeEventListener('nex:new-conversation', newHandler);
   150→    };
   151→  }, []);
   152→
   153→  // ── Phase 33: Startup restore (last conversation) ──
   154→  useEffect(() => {
   155→    (async () => {
   156→      try {

   158→        if (r.success && r.conversations && r.conversations.length > 0) {
   159→          const last = r.conversations[0]; // sorted by updatedAt desc
   160→          const load = await window.nexAPI.conversationLoad(last.id);
   161→          if (load.success && load.data) {
   162→            const validated = validateConversationData(load.data);
   163→            if (validated === null) {
   164→              console.warn('[NEX AI] Malformed conversation data on startup — starting fresh');
   165→              return;
   166→            }

   168→            setConversationId(load.data.id);
   169→            setConversationTitle(load.data.title || '');
   170→            conversationIdRef.current = load.data.id;
   171→            conversationTitleRef.current = load.data.title || '';
   172→            conversationCreatedAtRef.current = load.data.createdAt || null;
   173→            setLastSavedAt(Date.now());

   177→    })();
   178→  }, []); // run once on mount
   179→
   180→  // ── Phase 33: Keyboard shortcuts (not in input/textarea) ──

   184→      const isInput = target.tagName === 'INPUT' || target.tagName === 'TEXTAREA' || target.isContentEditable;

   198→

   205→          id: conversationIdRef.current,
   206→          title: conversationTitleRef.current || 'Untitled',

   222→  // ── Phase 33: Edit user message ──
   223→  const handleEditMessage = useCallback(async (messageId: string, newContent: string) => {
   224→    if (isGenerating || isResendingRef.current) return;
   225→    isResendingRef.current = true;
   226→    setTimeout(() => { isResendingRef.current = false; }, 1000);
   227→    setEditingMessageId(null);
   228→
   229→    const msgIndex = messages.findIndex((m) => m.id === messageId);
   230→    if (msgIndex === -1 || messages[msgIndex].role !== 'user') return;
   231→
   232→    // Truncate everything after the edited message
   233→    const updated = messages.slice(0, msgIndex);
   234→    const editedMsg = { ...messages[msgIndex], content: newContent, timestamp: Date.now() };
   235→    const newMessages = [...updated, editedMsg];
   236→    setMessages(newMessages);
   237→
   238→    // Re-send the edited message (will generate new assistant response)
   239→    setInput(newContent);
   240→    // Trigger send after state flush
   241→    setTimeout(() => {
   242→      const el = document.querySelector<HTMLTextAreaElement>('textarea[data-chat-input]');
   243→      if (el) {
   244→        el.value = newContent;
   245→        el.dispatchEvent(new Event('input', { bubbles: true }));
   246→        setTimeout(() => {
   247→          el.dispatchEvent(new KeyboardEvent('keydown', { key: 'Enter', bubbles: true }));
   248→        }, 50);
   249→      }
   250→    }, 10);
   251→
   252→    // Persist the truncated conversation
   253→    saveConversation(newMessages);
   254→  }, [messages, isGenerating, saveConversation]);
   255→
   256→  // ── Phase 33: Regenerate last assistant response ──
   257→  const handleRegenerate = useCallback(async () => {
   258→    if (isGenerating || isResendingRef.current) return;
   259→    isResendingRef.current = true;
   260→    setTimeout(() => { isResendingRef.current = false; }, 1000);
   261→
   262→    // Find last assistant message
   263→    const lastAssistantIdx = messages.findIndex((m) => m.role === 'assistant' && m === [...messages].reverse().find((r) => r.role === 'assistant'));
   264→    if (lastAssistantIdx === -1) return;
   265→
   266→    // Find the user message before it
   267→    let lastUserIdx = -1;
   268→    for (let i = lastAssistantIdx - 1; i >= 0; i--) {
   269→      if (messages[i].role === 'user') { lastUserIdx = i; break; }
   270→    }
   271→    if (lastUserIdx === -1) return;
   272→
   273→    // Remove the assistant response (keep everything before it)
   274→    const truncated = messages.slice(0, lastAssistantIdx);
   275→    setMessages(truncated);
   276→
   277→    // Re-send the same user message
   278→    const userContent = messages[lastUserIdx].content;
   279→    setInput(userContent);
   280→    setTimeout(() => {
   281→      const el = document.querySelector<HTMLTextAreaElement>('textarea[data-chat-input]');
   282→      if (el) {
   283→        el.value = userContent;
   284→        el.dispatchEvent(new Event('input', { bubbles: true }));
   285→        setTimeout(() => {
   286→          el.dispatchEvent(new KeyboardEvent('keydown', { key: 'Enter', bubbles: true }));
   287→        }, 50);
   288→      }
   289→    }, 10);
   290→  }, [messages, isGenerating]);
   291→
   292→  const startEdit = useCallback((messageId: string) => {
   293→    const msg = messages.find((m) => m.id === messageId);
   294→    if (msg?.role === 'user') {
   295→      setEditingMessageId(messageId);
   296→      setEditText(msg.content);
   297→    }
   298→  }, [messages]);
   299→
   300→  const cancelEdit = useCallback(() => {
   301→    setEditingMessageId(null);
   302→    setEditText('');
   303→  }, []);
   304→
   305→  const messagesEndRef = useRef<HTMLDivElement>(null);
   306→  const streamBufRef = useRef<string>('');
   307→  const fileInputRef = useRef<HTMLInputElement>(null);
   308→  const inputRef = useRef<HTMLTextAreaElement>(null);
   309→
   310→  // Phase 14: Track whether the current request originated from voice.
   311→  // Set to true when a nex:voice-transcript event with source='voice' arrives.
   312→  // Reset to false when keyboard/text input is submitted.
   313→  // Used after response completion to decide whether to speak via TTS.
   314→  const wasVoiceInputRef = useRef<boolean>(false);
   315→  // Phase 14: Track whether the current request was cancelled before TTS.
   316→  // Prevents stale requests from triggering TTS after cancellation.
   317→  const ttsCancelledRef = useRef<boolean>(false);
   318→
   319→  // ── Phase 14: Voice transcript listener (final transcripts → sendMessage) ──
   320→  // When a voice transcript arrives with source='voice', we set wasVoiceInputRef
   321→  // so the response will be spoken via TTS after completion.
   322→  useEffect(() => {
   323→    const handler = (e: Event) => {
   324→      const detail = (e as CustomEvent).detail;
   325→      if (detail?.text?.trim()) {
   326→        // Phase 14: Track voice origin for TTS
   327→        wasVoiceInputRef.current = detail.source === 'voice';
   328→        ttsCancelledRef.current = false; // reset cancellation flag for new request
   329→        // Feed into the SAME pipeline as typed input
   330→        setInput(detail.text.trim());
   331→        // Trigger send on next tick (input state needs to flush)
   332→        setTimeout(() => {
   333→          const el = document.querySelector<HTMLTextAreaElement>('textarea[data-chat-input]');
   334→          if (el) {
   335→            el.value = detail.text.trim();
   336→            el.dispatchEvent(new Event('input', { bubbles: true }));
   337→            // Small delay for state flush, then simulate Enter
   338→            setTimeout(() => {
   339→              el.dispatchEvent(new KeyboardEvent('keydown', { key: 'Enter', bubbles: true }));
   340→            }, 50);
   341→          }
   342→        }, 10);
   343→      }
   344→    };
   345→    window.addEventListener('nex:voice-transcript', handler);
   346→    return () => window.removeEventListener('nex:voice-transcript', handler);
   347→  }, []);
   348→
   349→  // Phase 110: Track active agent task for session stickiness
   350→  const activeAgentTaskRef = useRef<string | null>(null);
   351→
   352→  // Phase 18 (P2-1): Stale-timer protection for 'agent' + 'chat' condition
   353→  // auto-clears. Previously the 1500ms setTimeout timers were anonymous —
   354→  // their IDs were not captured. A stale timer from an OLD task could fire
   355→  // during a NEW task and clear the NEW task's condition (brief Orb blip).
   356→  // Now we capture the timer ID and clear the PREVIOUS timer before
   357→  // scheduling a new one for the SAME condition key. The invariant is:
   358→  //   A stale timer from an old task must NEVER clear the condition
   359→  //   belonging to a newer task.
   360→  const agentClearTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   361→  const chatClearTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   362→
   363→  // Phase 18 (P2-1): Helper to schedule a 1500ms auto-clear for a condition,
   364→  // clearing any previously-scheduled timer for the SAME key first.
   365→  const scheduleConditionClear = useCallback(
   366→    (key: 'agent' | 'chat', timerRef: React.MutableRefObject<ReturnType<typeof setTimeout> | null>) => {
   367→      // Clear any previously-scheduled timer for this condition key.
   368→      // This prevents a stale timer from an old task/event from clearing
   369→      // the NEW task's condition.
   370→      if (timerRef.current !== null) {
   371→        clearTimeout(timerRef.current);
   372→        timerRef.current = null;
   373→      }
   374→      // Schedule the new 1500ms auto-clear, capturing the timer ID.
   375→      timerRef.current = setTimeout(() => {
   376→        voiceController.clearCondition(key);
   377→        timerRef.current = null; // clear the ref so we know no timer is pending
   378→      }, 1500);
   379→    },
   380→    []
   381→  );
   382→
   383→  // Phase 14: Speak a response via TTS if the request came from voice.
   384→  // Guards:
   385→  //   - Only speaks if wasVoiceInputRef is true AND not cancelled
   386→  //   - Only speaks non-empty text (skips empty/error/intermediate)
   387→  //   - Best-effort: TTS failure doesn't crash the UI
   388→  //   - Resets wasVoiceInputRef after speaking (one-shot)
   389→  // Security: only the final assistant response text is passed to TTS.
   390→  // No API keys, tool params, or intermediate reasoning are spoken.
   391→  const speakResponseIfVoice = useCallback((responseText: string) => {
   392→    if (!wasVoiceInputRef.current) return; // text input → no TTS
   393→    if (ttsCancelledRef.current) return;  // cancelled → no stale TTS
   394→    const text = typeof responseText === 'string' ? responseText.trim() : '';
   395→    if (!text) return; // empty response → no TTS
   396→
   397→    // Reset voice flag (one-shot — next text request won't speak)
   398→    wasVoiceInputRef.current = false;
   399→
   400→    // Best-effort TTS — errors logged but don't crash UI
   401→    window.nexAPI?.voiceConversationSpeak?.(text).catch((err: any) => {
   402→      console.warn('[TTS] voiceConversationSpeak failed (non-blocking):', err?.message || err);
   403→    });
   404→  }, []);
   405→
   406→  // ── Phase 30: Voice thinking state (tell Orb when AI is processing) ──
   407→  useEffect(() => {
   408→    voiceController.setThinking(isGenerating);
   409→    return () => voiceController.setThinking(false);
   410→  }, [isGenerating]);
   411→
   412→  // Phase 116 JARVIS: Drive Orb state from agent events
   413→  // WORKING: when agent starts executing tools (step_started, tool_call_started)
   414→  // SUCCESS: when task completes successfully (task_completed)
   415→  // CANCELLED: when user cancels (task_cancelled)
   416→  // THINKING: when planning starts (planning_started)
   417→  // This runs INSIDE the agent event listener (below) via voiceController.setCondition
   418→
   419→  // ── Streaming listener ──
   420→  useEffect(() => {
   421→    const off = window.nexAPI.onChatToken((ev) => {
   422→      if (ev.text) {
   423→        streamBufRef.current += ev.text;
   424→        // Update the streaming message in place
   425→        setMessages((prev) => {
   426→          const next = [...prev];
   427→          const last = next[next.length - 1];
   428→          if (last && last.status === 'streaming' || last?.status === 'pending') {
   429→            next[next.length - 1] = { ...last, content: streamBufRef.current, status: 'streaming' };
   430→          }
   431→          return next;
   432→        });
   433→      }
   434→    });
   435→    return off;
   436→  }, []);
   437→
   438→  // ── Phase 109/110: Agent event listener — updates chat UI with agent progress ──
   439→  useEffect(() => {
   440→    const off = window.nexAPI?.onAgentEvent?.((event: any) => {
   441→      const eventType = event?.type || event?.event;
   442→      const taskId = event?.taskId;
   443→
   444→      setMessages((prev) => {
   445→        const next = [...prev];
   446→        const last = next[next.length - 1];
   447→        if (!last || last.metadata?.agentTaskId !== taskId) return prev;
   448→
   449→        switch (eventType) {
   450→          case 'planning_started':
   451→            // Phase 116 JARVIS: Orb THINKING while planning (LLM decision-making)
   452→            voiceController.setCondition('agent', 'thinking');
   453→            next[next.length - 1] = { ...last, content: '🧠 Agent is thinking...\n\n📋 Planning approach...' };
   454→            break;
   455→          case 'planning_completed':
   456→          case 'plan_created':
   457→            // Phase 116 JARVIS: Orb WORKING when tools start executing
   458→            voiceController.setCondition('agent', 'working');
   459→            next[next.length - 1] = { ...last, content: '🧠 Agent is working...\n\n📋 Plan created. Executing steps...' };
   460→            break;
   461→          case 'step_started':
   462→          case 'tool_call':
   463→          case 'tool_call_started':
   464→            // Phase 116 JARVIS: Orb WORKING (actual tool execution — NOT thinking)
   465→            // THINKING = LLM decision-making / planning
   466→            // WORKING = tool execution (file ops, commands, etc.)
   467→            voiceController.setCondition('agent', 'working');
   468→            next[next.length - 1] = {
   469→              ...last,
   470→              content: `🧠 Agent is working...\n\n🔧 Running ${event.toolName || event.step?.toolName || 'tool'}...`,
   471→            };
   472→            break;
   473→          case 'step_completed':
   474→          case 'tool_result':
   475→            // Phase 115: Capture snapshotId from tool_call_completed for Undo UI.
   476→            // The snapshotId is only present for file-modifying tools (write_file,
   477→            // edit_file). We store it in the message metadata so MessageBubble
   478→            // can show an Undo button.
   479→            {
   480→              const snapId = event?.data?.snapshotId;
   481→              const fileLabel = event?.data?.fileLabel;
   482→              if (snapId) {
   483→                next[next.length - 1] = {
   484→                  ...last,
   485→                  content: `🧠 Agent is working...\n\n✅ ${event.toolName || event.data?.toolName || 'Step'} completed.`,
   486→                  metadata: {
   487→                    ...last.metadata,
   488→                    snapshotId: snapId,
   489→                    fileLabel: fileLabel,
   490→                  },
   491→                };
   492→              } else {
   493→                next[next.length - 1] = {
   494→                  ...last,
   495→                  content: `🧠 Agent is working...\n\n✅ ${event.toolName || 'Step'} completed.`,
   496→                };
   497→              }
   498→            }
   499→            break;
   500→          case 'verification':
   501→            next[next.length - 1] = { ...last, content: '🧠 Agent is working...\n\n🔍 Verifying results...' };
   502→            break;
   503→          // Phase 9: explicit verification pass/fail events
   504→          case 'verification_passed':
   505→            next[next.length - 1] = {
   506→              ...last,

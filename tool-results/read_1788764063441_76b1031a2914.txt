   172→function formatBytes(n: number): string {
   173→  if (!n) return '0 B';
   174→  if (n < 1024) return `${n} B`;
   175→  if (n < 1024 * 1024) return `${(n / 1024).toFixed(0)} KB`;
   176→  if (n < 1024 * 1024 * 1024) return `${(n / (1024 * 1024)).toFixed(0)} MB`;
   177→  return `${(n / (1024 * 1024 * 1024)).toFixed(2)} GB`;
   178→}

   180→function AIStorageSection() {
   181→  const [storageInfo, setStorageInfo] = useState<any>(null);
   182→  const [scanning, setScanning] = useState(false);
   183→  const [repairing, setRepairing] = useState(false);
   184→  const [toast, setToast] = useState<string | null>(null);
   185→
   186→  const refresh = useCallback(async () => {
   187→    try {
   188→      const res = await window.nexAPI.aiStorageInfo();
   189→      if (res.success) setStorageInfo(res);
   190→    } catch {}
   191→  }, []);
   192→
   193→  useEffect(() => { refresh(); }, [refresh]);
   194→
   195→  const showToast = (msg: string) => { setToast(msg); setTimeout(() => setToast(null), 3000); };
   196→
   197→  const handleChooseFolder = async () => {
   198→    const res = await window.nexAPI.aiStorageChooseFolder();
   199→    if (res.success) {
   200→      showToast('Storage location updated');
   201→      refresh();
   202→    } else if (!res.cancelled) {
   203→      showToast('Error: ' + (res.error || 'failed'));
   204→    }
   205→  };
   206→
   207→  const handleScan = async () => {
   208→    setScanning(true);
   209→    try {
   210→      const res = await window.nexAPI.aiStorageScan();
   211→      if (res.success) {
   212→        showToast(`Scan complete: ${res.registered} new, ${res.alreadyRegistered} existing`);
   213→        refresh();
   214→      } else {
   215→        showToast('Scan failed: ' + (res.error || 'unknown'));
   216→      }
   217→    } catch (err: any) {
   218→      showToast('Error: ' + err?.message);
   219→    } finally {
   220→      setScanning(false);
   221→    }
   222→  };
   223→
   224→  const handleRepair = async () => {
   225→    setRepairing(true);
   226→    try {
   227→      const res = await window.nexAPI.aiStorageRepair();
   228→      if (res.success) {
   229→        showToast(`Registry repaired: ${res.removed} removed, ${res.total} valid`);
   230→        refresh();
   231→      }
   232→    } catch {} finally {
   233→      setRepairing(false);
   234→    }
   235→  };
   236→
   237→  const handleOpenFolder = async () => {
   238→    await window.nexAPI.aiStorageOpenFolder();
   239→  };
   240→
   241→  if (!storageInfo) {
   242→    return <div className="text-xs" style={{ color: 'var(--nex-text-muted)' }}>Loading...</div>;
   243→  }
   244→
   245→  return (
   246→    <div className="space-y-3">
   247→      {/* Current path */}
   248→      <Row label="Path" value={storageInfo.path} mono />
   249→
   250→      {/* Stats */}
   251→      <div className="grid grid-cols-2 gap-2 py-1">
   252→        <div className="p-2 rounded-md" style={{ background: 'var(--nex-glass-bg)', border: '1px solid var(--nex-glass-border)' }}>
   253→          <div className="text-[10px]" style={{ color: 'var(--nex-text-muted)' }}>Total Size</div>
   254→          <div className="text-sm font-mono" style={{ color: 'var(--nex-text)' }}>{formatBytes(storageInfo.totalSize || 0)}</div>
   255→        </div>
   256→        <div className="p-2 rounded-md" style={{ background: 'var(--nex-glass-bg)', border: '1px solid var(--nex-glass-border)' }}>
   257→          <div className="text-[10px]" style={{ color: 'var(--nex-text-muted)' }}>Models</div>
   258→          <div className="text-sm font-mono" style={{ color: 'var(--nex-text)' }}>{storageInfo.modelCount || 0}</div>
   259→        </div>
   260→        <div className="p-2 rounded-md" style={{ background: 'var(--nex-glass-bg)', border: '1px solid var(--nex-glass-border)' }}>
   261→          <div className="text-[10px]" style={{ color: 'var(--nex-text-muted)' }}>Voice Components</div>
   262→          <div className="text-sm font-mono" style={{ color: 'var(--nex-text)' }}>{storageInfo.voiceCount || 0}</div>
   263→        </div>
   264→        <div className="p-2 rounded-md" style={{ background: 'var(--nex-glass-bg)', border: '1px solid var(--nex-glass-border)' }}>
   265→          <div className="text-[10px]" style={{ color: 'var(--nex-text-muted)' }}>Documents</div>
   266→          <div className="text-sm font-mono" style={{ color: 'var(--nex-text)' }}>{storageInfo.documentCount || 0}</div>
   267→        </div>
   268→      </div>
   269→
   270→      {/* Buttons */}
   271→      <div className="flex flex-wrap gap-2">
   272→        <ActionButton onClick={handleChooseFolder}>
   273→          <span className="flex items-center gap-1"><HardDrive size={11} /> Change Location</span>
   274→        </ActionButton>
   275→        <ActionButton onClick={handleScan}>
   276→          <span className="flex items-center gap-1">
   277→            {scanning ? <RefreshCw size={11} className="animate-spin" /> : <RefreshCw size={11} />}
   278→            Scan Storage
   279→          </span>
   280→        </ActionButton>
   281→        <ActionButton onClick={handleRepair}>
   282→          <span className="flex items-center gap-1">
   283→            {repairing ? <RefreshCw size={11} className="animate-spin" /> : <Shield size={11} />}
   284→            Repair Registry
   285→          </span>
   286→        </ActionButton>
   287→        <ActionButton onClick={handleOpenFolder}>
   288→          <span className="flex items-center gap-1"><Globe size={11} /> Open Folder</span>
   289→        </ActionButton>
   290→      </div>
   291→
   292→      {/* Toast */}
   293→      {toast && (
   294→        <div className="p-2 rounded-md text-xs" style={{ background: 'rgba(6,182,212,0.1)', color: '#67e8f9', border: '1px solid rgba(6,182,212,0.2)' }}>
   295→          {toast}
   296→        </div>
   297→      )}
   298→
   299→      {/* Help text */}
   300→      <div className="text-[10px] p-2 rounded" style={{ color: 'var(--nex-text-muted)', background: 'var(--nex-glass-bg)' }}>
   301→        Download models manually, place them in the correct folder (e.g. models/llm/qwen/),
   302→        then click Scan Storage. NEX AI will automatically detect, classify, and register them.
   303→      </div>
   304→    </div>
   305→  );
   306→}
